function drawGraph(v) { const canvas = document.getElementById("graph"); if (!canvas) return; const ctx = canvas.getContext("2d"); // Use the actual canvas resolution const W = canvas.width; const H = canvas.height; ctx.clearRect(0, 0, W, H); ctx.fillStyle = "#ffffff"; ctx.fillRect(0, 0, W, H); const left = 85; const right = W - 25; const top = 25; const bottom = H - 65; const plotWidth = right - left; const plotHeight = bottom - top; // Fixed positive ranges prevent a zero-size graph const maxStrain = Math.max(v.strain * 1.25, 0.01); const maxStress = Math.max(v.stress * 1.25, 1); // Draw grid lines ctx.strokeStyle = "#d6e7f7"; ctx.lineWidth = 1; for (let i = 0; i <= 5; i++) { const x = left + (plotWidth * i) / 5; const y = top + (plotHeight * i) / 5; ctx.beginPath(); ctx.moveTo(x, top); ctx.lineTo(x, bottom); ctx.stroke(); ctx.beginPath(); ctx.moveTo(left, y); ctx.lineTo(right, y); ctx.stroke(); } // Draw axes ctx.strokeStyle = "#173c70"; ctx.lineWidth = 2; ctx.beginPath(); ctx.moveTo(left, top); ctx.lineTo(left, bottom); ctx.lineTo(right, bottom); ctx.stroke(); // Axis numbers ctx.fillStyle = "#173c70"; ctx.font = "12px Arial"; for (let i = 0; i <= 5; i++) { const strainValue = maxStrain * i / 5; const stressValue = maxStress * i / 5; const x = left + plotWidth * i / 5; const y = bottom - plotHeight * i / 5; ctx.textAlign = "center"; ctx.fillText(strainValue.toExponential(1), x, bottom + 20); ctx.textAlign = "right"; ctx.fillText(stressValue.toExponential(1), left - 9, y + 4); } // X-axis label ctx.textAlign = "center"; ctx.font = "bold 14px Arial"; ctx.fillText( "Shear Strain (γ)", (left + right) / 2, H - 15 ); // Y-axis label ctx.save(); ctx.translate(20, (top + bottom) / 2); ctx.rotate(-Math.PI / 2); ctx.textAlign = "center"; ctx.fillText("Shear Stress (Pa)", 0, 0); ctx.restore(); // Convert a data point into canvas coordinates function graphX(strain) { return left + (strain / maxStrain) * plotWidth; } function graphY(stress) { return bottom - (stress / maxStress) * plotHeight; } // Draw the stress-strain line from the origin ctx.strokeStyle = "#087df0"; ctx.lineWidth = 3; ctx.beginPath(); ctx.moveTo(graphX(0), graphY(0)); ctx.lineTo(graphX(v.strain), graphY(v.stress)); ctx.stroke(); // Draw the current reading ctx.fillStyle = "#ff8a00"; ctx.beginPath(); ctx.arc( graphX(v.strain), graphY(v.stress), 6, 0, Math.PI * 2 ); ctx.fill(); // Graph legend ctx.fillStyle = "#087df0"; ctx.fillRect(right - 145, top + 8, 14, 4); ctx.fillStyle = "#173c70"; ctx.font = "12px Arial"; ctx.textAlign = "right"; ctx.fillText("Current reading", right, top + 27); }
